import React from 'react';
function App() {
    function showDate() {
        alert(new Date().toDateString());
    }
    function showByr() {
        alert("Hello! Welcome");
    }
    return (
        <div>
            <h2>Event Handling</h2>
            <button onClick={showDate}>Show Date</button>
            <button onClick={showByr}>Show Message</button>
        </div>
    );
}
export default App;


import React, { useState } from 'react';
function App() {
  const [message, setMessage] = useState("");
  function showDate() {
    setMessage(new Date().toLocaleString());
  }
  function showBye() {
    setMessage("Good Bye");
  }
  return (
    <div>
      <h1>Event Handling</h1>
      <button onClick={showDate}>Show Date and Time</button>      
      <br /><br />
      <button onClick={showBye}>Show Bye</button>
      
      {/* Render the message state */}
      <h2>{message}</h2>
    </div>
  );
}
export default App;



import React, { Component } from 'react';
class App extends Component {
  constructor() {
    super();
    this.state = { count: 0 };
  }
  componentDidMount() {
    console.log("Component Mounted");
  }
  componentDidUpdate() {
    console.log("Component Updated");
  }
  render() {
    return (
      <div>
        <h1>React Lifecycle</h1>
        <h2>Count: {this.state.count}</h2>
        <button onClick={() => this.setState({ count: this.state.count + 1 })}>
          Increase
        </button>
      </div>
    );
  }
}
export default App;
