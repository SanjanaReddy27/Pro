const express = require('express');
const app = express();

app.use(express.json());

let students = [];

// Create
app.post('/add', (req, res) => {
    students.push(req.body);
    res.send("Added");
});

// Read
app.get('/view', (req, res) => {
    res.json(students);
});

// Update
app.put('/update', (req, res) => {
    students[0] = req.body;
    res.send("Updated");
});

// Delete
app.delete('/delete', (req, res) => {
    students.pop();
    res.send("Deleted");
});

app.listen(3000);
